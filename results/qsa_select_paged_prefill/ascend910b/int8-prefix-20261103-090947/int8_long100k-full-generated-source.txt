import triton
import triton.language as tl

@triton.jit
def _quant_score_trial(Q, K, QS, KS, Weights, Table, Visible, Out, RawDot, HEADS: tl.constexpr, PAGE: tl.constexpr, COLUMNS: tl.constexpr, ROWS: tl.constexpr, WIDTH: tl.constexpr, BLOCK_N: tl.constexpr, MASK_RAW_DOT: tl.constexpr, WRITE_RAW_DOT: tl.constexpr):
    tiles: tl.constexpr = triton.cdiv(WIDTH, BLOCK_N)
    dims = tl.arange(0, 128)
    lanes = tl.arange(0, 16)
    for work in range(tl.program_id(0), ROWS * tiles, tl.num_programs(0)):
        row = work // tiles
        columns = work % tiles * BLOCK_N + tl.arange(0, BLOCK_N)
        visible = tl.load(Visible + row)
        live = (columns < WIDTH) & (columns < visible)
        physical = tl.load(Table + row * COLUMNS + columns // PAGE, mask=live, other=0)
        token = physical.to(tl.int64) * PAGE + columns % PAGE
        keys = tl.load(K + token[:, None] * 128 + dims[None, :], mask=live[:, None], other=0)
        key_scale = tl.load(KS + token, mask=live, other=0).to(tl.float32)
        scores = tl.full((BLOCK_N,), 0, tl.float32)
        for group in tl.static_range(triton.cdiv(HEADS, 16)):
            heads = group * 16 + lanes
            query = tl.load(Q + (row * HEADS + heads[None, :]) * 128 + dims[:, None], mask=heads[None, :] < HEADS, other=0)
            dot = tl.dot(keys, query, out_dtype=tl.int32)
            if WRITE_RAW_DOT:
                if MASK_RAW_DOT:
                    debug_dot = tl.where(live[:, None], dot, 0)
                else:
                    debug_dot = dot
                tl.store(RawDot + (row * WIDTH + columns[:, None]) * HEADS + heads[None, :], debug_dot, mask=(columns[:, None] < WIDTH) & (heads[None, :] < HEADS))
            query_scale = tl.load(QS + row * HEADS + heads, mask=heads < HEADS, other=0).to(tl.float32)
            weights = tl.load(Weights + row * HEADS + heads, mask=heads < HEADS, other=0).to(tl.float32)
            dequant = dot.to(tl.float32) * query_scale[None, :] * key_scale[:, None]
            scores += tl.sum(tl.maximum(dequant, 0.0) * weights[None, :], axis=1)
        tl.store(Out + row * WIDTH + columns, tl.where(live, scores, -float('inf')), mask=columns < WIDTH)

import triton
import triton.language as tl

@triton.jit
def raw_finish_merge256(Raw, KS, Table, Visible, Out, ROWS:tl.constexpr,
                            WIDTH:tl.constexpr,RAW_TILES,CORES:tl.constexpr):
    for unit in range(tl.program_id(0),ROWS*triton.cdiv(WIDTH,256),CORES):
        row = unit//triton.cdiv(WIDTH,256)
        block = unit%triton.cdiv(WIDTH,256)
        tiles = block*4+tl.arange(0,4)
        c = tl.arange(0,64)
        visible = tl.load(Visible+row)
        segment_live = (tiles*64<WIDTH)&(tiles*64<visible)
        page = tl.where(segment_live,tiles//3,0)
        physical = tl.load(Table+page,mask=segment_live,other=0)
        safe_tile = tl.where(segment_live,tiles,0)
        live = (tiles[:,None]*64+c[None,:]<WIDTH)&(tiles[:,None]*64+c[None,:]<visible)
        ks = tl.load(KS+physical[:,None].to(tl.int64)*192+safe_tile[:,None]%3*64+c[None,:],
                     mask=live,other=0).to(tl.float32)
        value = tl.load(Raw+((row//16)*RAW_TILES+safe_tile[:,None])*1024+row%16*64+c[None,:],
                        mask=live,other=0)*ks
        result = tl.reshape(tl.where(live,value,-float("inf")),(256,))
        out_col = block*256+tl.arange(0,256)
        tl.store(Out+row*WIDTH+out_col,result,mask=out_col<WIDTH)

@triton.jit
def raw_full_tail(Raw, KS, Table, Out, row, block, visible,

    tiles = block*16+tl.arange(0,16)
    c = tl.arange(0,64)

    safe_tile = tl.where(segment_live,tiles,0)
    live = (tiles[:,None]*64+c[None,:]<WIDTH)&(tiles[:,None]*64+c[None,:]<visible)

    tl.store(Out+row*WIDTH+out_col,result,mask=out_col<WIDTH)

@triton.jit
def raw_full_body(Raw, KS, Table, Out, row, block, WIDTH:tl.constexpr,RAW_TILES):
    tiles = block*16+tl.arange(0,16)
    c = tl.arange(0,64)
    physical = tl.load(Table+tiles//3)
    ks = tl.load(KS+physical[:,None].to(tl.int64)*192+tiles[:,None]%3*64+c[None,:]).to(tl.float32)
    value = tl.load(Raw+((row//16)*RAW_TILES+tiles[:,None])*1024+row%16*64+c[None,:])*ks
    result = tl.reshape(value,(1024,))
    col = block*1024+tl.arange(0,1024)
    tl.store(Out+row*WIDTH+col,result)

@triton.jit
def raw_finish_full1024(Raw, KS, Table, Visible, Out, ROWS:tl.constexpr,
                        WIDTH:tl.constexpr,RAW_TILES,CORES:tl.constexpr):
    for unit in range(tl.program_id(0),ROWS*triton.cdiv(WIDTH,1024),CORES):
        row = unit//triton.cdiv(WIDTH,1024)
        block = unit%triton.cdiv(WIDTH,1024)
        visible = tl.load(Visible+row)
        if ((block+1)*1024<=WIDTH)&((block+1)*1024<=visible):
            raw_full_body(Raw,KS,Table,Out,row,block,WIDTH,RAW_TILES)
        else:
            raw_full_tail(Raw,KS,Table,Out,row,block,visible,WIDTH,RAW_TILES)

import triton
import triton.language as tl

@triton.jit
def raw_k_2d_part(K, Table, PK, part, MAX_VISIBLE):
    tile = part//8
    row_start = part%2*32
    dim_start = (part%8//2)*32
    r = tl.arange(0,32)
    c = tl.arange(0,32)
    physical = tl.load(Table+tile//3)
    token = physical.to(tl.int64)*192+tile%3*64+row_start
    global_row = tile*64+row_start+r
    value = tl.load(K+(token+r[:,None])*128+dim_start+c[None,:],
                    mask=global_row[:,None]<MAX_VISIBLE,other=0)
    tl.store(PK+part*1024+r[:,None]*32+c[None,:],value)

@triton.jit
def raw_k_2d_grid(K, Table, PK, MAX_VISIBLE):
    raw_k_2d_part(K,Table,PK,tl.program_id(0),MAX_VISIBLE)

@triton.jit
def raw_k_2d_capped(K, Table, PK, MAX_VISIBLE,TILES,CORES:tl.constexpr):
    for part in range(tl.program_id(0),TILES*8,CORES):
        raw_k_2d_part(K,Table,PK,part,MAX_VISIBLE)

@triton.jit
def raw_pack_k_ks(K, KS, Table, PK, LogicalKS, MAX_VISIBLE,TILES,CORES:tl.constexpr):
    for part in range(tl.program_id(0),TILES*8,CORES):
        raw_k_2d_part(K,Table,PK,part,MAX_VISIBLE)
        if part%8 == 0:
            tile = part//8
            c = tl.arange(0,64)
            physical = tl.load(Table+tile//3)
            scale = tl.load(KS+physical.to(tl.int64)*192+tile%3*64+c,
                            mask=tile*64+c<MAX_VISIBLE,other=0)
            tl.store(LogicalKS+tile*64+c,scale)

@triton.jit
def raw_logical_body(Raw, LogicalKS, Out, row, block, WIDTH:tl.constexpr,RAW_TILES):
    tiles = block*16+tl.arange(0,16)
    c = tl.arange(0,64)
    col = block*1024+tl.arange(0,1024)
    ks = tl.reshape(tl.load(LogicalKS+col).to(tl.float32),(16,64))
    value = tl.load(Raw+((row//16)*RAW_TILES+tiles[:,None])*1024+row%16*64+c[None,:])*ks
    tl.store(Out+row*WIDTH+col,tl.reshape(value,(1024,)))

@triton.jit
def raw_finish_logical1024(Raw, KS, Table, Visible, Out, LogicalKS, ROWS:tl.constexpr,
                           WIDTH:tl.constexpr,RAW_TILES,CORES:tl.constexpr):
    for unit in range(tl.program_id(0),ROWS*triton.cdiv(WIDTH,1024),CORES):
        row = unit//triton.cdiv(WIDTH,1024)
        block = unit%triton.cdiv(WIDTH,1024)
        visible = tl.load(Visible+row)
        if ((block+1)*1024<=WIDTH)&((block+1)*1024<=visible):
            raw_logical_body(Raw,LogicalKS,Out,row,block,WIDTH,RAW_TILES)
        else:
            raw_full_tail(Raw,KS,Table,Out,row,block,visible,WIDTH,RAW_TILES)

@triton.jit
def raw_query16_group(Q, QS, W, PQ, PW, group, ROWS:tl.constexpr, QUERIES:tl.constexpr):
    heads = tl.arange(0,16)
    HEADS:tl.constexpr = QUERIES*4
    dims = tl.arange(0,32)
    for head_block in tl.static_range(QUERIES//4):
        flat = group*HEADS+heads+head_block*16
        for block in tl.static_range(4):
            query = tl.load(Q+flat[:,None]*128+block*32+dims[None,:],
                            mask=flat[:,None]<ROWS*4,other=0)
            tl.store(PQ+group*(HEADS*128)+block*(HEADS*32)+(heads[:,None]+head_block*16)*32+dims[None,:],query)
        qs = tl.load(QS+flat,mask=flat<ROWS*4,other=0).to(tl.float32)
        weights = tl.load(W+flat,mask=flat<ROWS*4,other=0).to(tl.float32)
        coefficient = (qs*weights*1024.0).to(tl.float16)
        out_rows = tl.arange(0,16)
        live = (out_rows[None,:]<QUERIES)&((heads[:,None]+head_block*16)//4==out_rows[None,:])
        # Native LoadData2D transposes W.T from L1 into the second Cube's L0A.
        matrix = tl.where(live,coefficient[:,None],tl.full((),0.0,tl.float16))
        tl.store(PW+group*(HEADS*16)+(heads[:,None]+head_block*16)*16+out_rows[None,:],matrix)


@triton.jit
def raw_query16_qw(Q, QS, W, PQ, PW, ROWS:tl.constexpr, QUERIES:tl.constexpr):
    raw_query16_group(Q,QS,W,PQ,PW,tl.program_id(0),ROWS,QUERIES)

@triton.jit
def raw_query16_cube(PQ, PK, PW, Raw, GROUPS:tl.constexpr,TILES,
                        CORES:tl.constexpr,QUERIES:tl.constexpr):
    with al.scope(core_mode="cube"):
        q_base = tl.cast(PQ,tl.uint64)
        key_base = tl.cast(PK,tl.uint64)
        weight_base = tl.cast(PW,tl.uint64)
        output_base = tl.cast(Raw,tl.uint64)

import triton
import triton.language as tl

@triton.jit
def raw_query16_finish_part(Raw, KS, Table, Visible, Out, row, tile,
                          WIDTH:tl.constexpr, RAW_TILES,QUERIES:tl.constexpr):
    c = tl.arange(0,64)
    first = tile*64
    col = first+c
    visible = tl.load(Visible+row)
    physical = tl.load(Table+tile//3,mask=(first<WIDTH)&(first<visible),other=0)
    key_base = physical.to(tl.int64)*192+tile%3*64
    raw_base = ((row//QUERIES)*RAW_TILES+tile)*1024+row%QUERIES*64
    live = (col<WIDTH)&(col<visible)
    ks = tl.load(KS+key_base+c,mask=live,other=0).to(tl.float32)
    value = tl.load(Raw+raw_base+c,mask=live,other=0)*ks
    tl.store(Out+row*WIDTH+col,tl.where(live,value,-float("inf")),mask=col<WIDTH)

@triton.jit
def raw_query16_finish_grid(Raw, KS, Table, Visible, Out,
                          WIDTH:tl.constexpr, RAW_TILES,QUERIES:tl.constexpr):
    raw_query16_finish_part(Raw,KS,Table,Visible,Out,tl.program_id(0),tl.program_id(1),
                          WIDTH,RAW_TILES,QUERIES)

@triton.jit
def raw_query16_finish_capped(Raw, KS, Table, Visible, Out, ROWS:tl.constexpr,
                            WIDTH:tl.constexpr, RAW_TILES,QUERIES:tl.constexpr,CORES:tl.constexpr):
    for unit in range(tl.program_id(0),ROWS*triton.cdiv(WIDTH,64),CORES):
        row = unit//triton.cdiv(WIDTH,64)
        tile = unit%triton.cdiv(WIDTH,64)
        raw_query16_finish_part(Raw,KS,Table,Visible,Out,row,tile,WIDTH,RAW_TILES,QUERIES)

@triton.jit
def raw_logical_tail(Raw, LogicalKS, Out, row, block, visible,
                     WIDTH:tl.constexpr, RAW_TILES):
    tiles = block*16+tl.arange(0,16)
    c = tl.arange(0,64)
    col = block*1024+tl.arange(0,1024)
    live = (tiles[:,None]*64+c[None,:]<WIDTH)&(tiles[:,None]*64+c[None,:]<visible)
    segment_live = (tiles*64<WIDTH)&(tiles*64<visible)
    safe_tile = tl.where(segment_live,tiles,0)
    ks_col = tl.reshape(tl.where(live,tiles[:,None]*64+c[None,:],0),(1024,))
    ks = tl.reshape(tl.load(LogicalKS+ks_col,mask=tl.reshape(live,(1024,)),other=0).to(tl.float32),(16,64))
    value = tl.load(Raw+((row//16)*RAW_TILES+safe_tile[:,None])*1024+row%16*64+c[None,:],
                    mask=live,other=0)*ks
    result = tl.reshape(tl.where(live,value,-float("inf")),(1024,))
    tl.store(Out+row*WIDTH+col,result,mask=col<WIDTH)

@triton.jit
def raw_empty_block(Out,row,block,WIDTH:tl.constexpr):
    col = block*1024+tl.arange(0,1024)
    tl.store(Out+row*WIDTH+col,tl.full((1024,),-float("inf"),tl.float32),mask=col<WIDTH)

@triton.jit
def raw_finish_empty1024(Raw,KS,Table,Visible,Out,LogicalKS,ROWS:tl.constexpr,
                      WIDTH:tl.constexpr,RAW_TILES,CORES:tl.constexpr):
    for unit in range(tl.program_id(0),ROWS*triton.cdiv(WIDTH,1024),CORES):
        row = unit//triton.cdiv(WIDTH,1024)
        block = unit%triton.cdiv(WIDTH,1024)
        visible = tl.load(Visible+row)
        if block*1024>=visible:

        elif ((block+1)*1024<=WIDTH)&((block+1)*1024<=visible):
            raw_logical_body(Raw,LogicalKS,Out,row,block,WIDTH,RAW_TILES)
        else:
            raw_full_tail(Raw,KS,Table,Out,row,block,visible,WIDTH,RAW_TILES)

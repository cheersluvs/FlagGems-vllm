  225| @builtin
  226| def fixpipe(
  227|     src: tl.tensor,
  228|     dst: bl.buffer,
  229|     dma_mode: FixpipeDMAMode = FixpipeDMAMode.NZ2ND,
  230|     dual_dst_mode: FixpipeDualDstMode = FixpipeDualDstMode.NO_DUAL,
  231|     _builder=None,
  232| ) -> None:
  233|     """
  234|     Directly store a tensor on L0C to a local buffer via fixpipe.
  235|     Fixpipe is pipeline that performing data movement from L0C to other memory hierarchies.
  236|     Currently support:
  237|         - L0C to UB (for Ascend910_95 sereies)
  238| 
  239|     :param src: the source tensor, Must be located in the l0C memory region.
  240|     :type src: tl.tensor
  241|     :param dst: The destination buffer, Must be located in the UB memory region.
  242|     :type dst: bl.buffer
  243|     :param dma_mode: DMA transfer mode, "nz2nd" enables NZ to ND layout transformation
  244|     :type dma_mode: str
  245|     """
  246|     if not _builder.is_910_95():
  247|         raise RuntimeError("this feature is only supported on Ascend910_95")
  248|     if not isinstance(src, tl.tensor):
  249|         raise TypeError("src is not of tensor type")
  250|     elif not isinstance(dst, bl.buffer):
  251|         raise TypeError("dst is not of buffer type")
  252|     if dst.space != ascend_address_space.UB:
  253|         raise TypeError("dst must be located in the UB memory region")
  254| 
  255|     if len(dst.shape) == 2 and (dst.type.element_ty == tl.float32 or dst.type.element_ty == tl.int32):
  256|         N = dst.shape[1]
  257|         if N % 8 != 0:
  258|             raise ValueError("32b Fixpipe last dim must be aligned to 8")
  259|         if (dma_mode != FixpipeDMAMode.NZ2ND) and (N % 16 != 0):
  260|             raise ValueError("32b non-NZ2ND Fixpipe last dim must be aligned to 16")
  261|         if (dual_dst_mode == FixpipeDualDstMode.COLUMN_SPLIT) and (N % 32 != 0):
  262|             raise ValueError("32b Column split dual Fixpipe last dim must be aligned to 32")
  263|         M = dst.shape[0]
  264|         if (dma_mode == FixpipeDMAMode.NZ2DN) and (M % 8 != 0):
  265|             raise ValueError("32b NZ2DN Fixpipe first dim must be aligned to 8")
  266|     dst16bits = (dst.type.element_ty == tl.float16 or dst.type.element_ty == tl.int16
  267|                  or dst.type.element_ty == tl.bfloat16)
  268|     if len(dst.shape) == 2 and dst16bits:
  269|         N = dst.shape[1]
  270|         if N % 16 != 0:
  271|             raise ValueError("16b Fixpipe last dim must be aligned to 16")
  272|         M = dst.shape[0]
  273|         if (dma_mode == FixpipeDMAMode.NZ2DN) and (M % 16 != 0):
  274|             raise ValueError("16b NZ2DN Fixpipe first dim must be aligned to 16")
  275| 
  276|     return semantic.fixpipe(src, dst, dma_mode, dual_dst_mode, FixpipePreQuantMode.NO_QUANT, FixpipePreReluMode.NO_RELU,
  277|                             _builder)
